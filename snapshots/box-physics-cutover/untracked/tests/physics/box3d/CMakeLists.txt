cmake_minimum_required(VERSION 3.22)
project(egp_box3d_tests LANGUAGES C CXX)
find_package(Threads REQUIRED)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BOX3D_AVX2 OFF CACHE BOOL "Fixed SIMD width" FORCE)
set(BOX3D_DOUBLE_PRECISION OFF CACHE BOOL "Qualified profile is float32" FORCE)
set(BOX3D_PROFILE OFF CACHE BOOL "No profiling dependency" FORCE)
if(MSVC)
  add_compile_options(/fp:precise)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    add_compile_options(/clang:-ffp-contract=off)
  endif()
else()
  add_compile_options(-fno-fast-math -ffp-contract=off)
endif()
add_subdirectory(../../../thirdparty/box3d box3d)
add_executable(egp_box3d_determinism determinism.cpp ../../../modules/box3d/deterministic_world.cpp)
target_compile_features(egp_box3d_determinism PRIVATE cxx_std_17)
target_include_directories(egp_box3d_determinism PRIVATE ../../../modules/box3d)
target_compile_definitions(egp_box3d_determinism PRIVATE EGP_BOX3D_GOLDEN_PATH="${CMAKE_CURRENT_SOURCE_DIR}/golden_hashes.txt")
target_link_libraries(egp_box3d_determinism PRIVATE box3d::box3d Threads::Threads)
enable_testing()
add_test(NAME egp_box3d_determinism COMMAND egp_box3d_determinism)
set_tests_properties(egp_box3d_determinism PROPERTIES TIMEOUT 90)
