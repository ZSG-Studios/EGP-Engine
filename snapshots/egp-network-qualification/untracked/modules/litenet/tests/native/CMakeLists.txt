cmake_minimum_required(VERSION 3.22)
project(egp_network_physics LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BOX3D_AVX2 OFF CACHE BOOL "Fixed SIMD width" FORCE)
set(BOX3D_DOUBLE_PRECISION OFF CACHE BOOL "Qualified float32 profile" FORCE)
set(BOX3D_PROFILE OFF CACHE BOOL "No profiling dependency" FORCE)
if(MSVC)
  add_compile_options(/fp:precise)
else()
  add_compile_options(-fno-fast-math -ffp-contract=off)
endif()
add_subdirectory(../../../../thirdparty/box3d box3d)
add_library(egp_network_physics SHARED physics_interop.cpp ../../../../modules/box3d/deterministic_world.cpp)
target_compile_features(egp_network_physics PRIVATE cxx_std_17)
target_include_directories(egp_network_physics PRIVATE ../../../../modules/box3d)
target_link_libraries(egp_network_physics PRIVATE box3d::box3d)
