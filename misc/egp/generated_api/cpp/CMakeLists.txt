cmake_minimum_required(VERSION 3.20)
project(superposition_api_probe LANGUAGES CXX)
if(NOT EGP_CPP_SDK)
  message(FATAL_ERROR "Pass -DEGP_CPP_SDK=<freshly extracted matching editor SDK>")
endif()
add_library(superposition_api_probe OBJECT probe.cpp)
target_include_directories(superposition_api_probe SYSTEM PRIVATE "${EGP_CPP_SDK}/include" "${EGP_CPP_SDK}/gen/include")
target_include_directories(superposition_api_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../..")
target_compile_features(superposition_api_probe PRIVATE cxx_std_17)
target_compile_definitions(superposition_api_probe PRIVATE GDEXTENSION THREADS_ENABLED TYPED_METHOD_BIND NOMINMAX DEBUG_ENABLED HOT_RELOAD_ENABLED)
target_compile_options(superposition_api_probe PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8;/Zc:__cplusplus;/W4;/WX>)
target_compile_options(superposition_api_probe PRIVATE $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-Wall;-Wextra;-Werror>)
