cmake_minimum_required(VERSION 3.20)
project(EGPNetChecks LANGUAGES C CXX)
enable_testing()
set(YOJIMBO_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(YOJIMBO_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../thirdparty/yojimbo yojimbo)
add_library(egp_net_core ../net_core.cpp)
find_package(Threads REQUIRED)
target_compile_features(egp_net_core PUBLIC cxx_std_17)
target_include_directories(egp_net_core PUBLIC ..)
target_link_libraries(egp_net_core PUBLIC yojimbo Threads::Threads)
if(MSVC)
  target_compile_options(egp_net_core PRIVATE /fp:precise)
else()
  target_compile_options(egp_net_core PRIVATE -ffp-contract=off)
endif()
add_executable(egp_net_checks core_checks.cpp)
target_link_libraries(egp_net_checks PRIVATE egp_net_core)
add_test(NAME egp_net_checks COMMAND egp_net_checks)
set_tests_properties(egp_net_checks PROPERTIES TIMEOUT 40)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_executable(egp_net_process_check process_check.cpp)
target_link_libraries(egp_net_process_check PRIVATE egp_net_core)
add_test(NAME egp_net_separate_processes COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/process_check.py $<TARGET_FILE:egp_net_process_check>)
set_tests_properties(egp_net_separate_processes PROPERTIES TIMEOUT 25)
add_executable(egp_net_interest_memory_check interest_memory_check.cpp)
target_link_libraries(egp_net_interest_memory_check PRIVATE egp_net_core)
add_test(NAME egp_net_interest_memory COMMAND egp_net_interest_memory_check)
set_tests_properties(egp_net_interest_memory PROPERTIES TIMEOUT 15)
add_executable(egp_net_state_pressure_check state_pressure_check.cpp)
target_link_libraries(egp_net_state_pressure_check PRIVATE egp_net_core)
add_test(NAME egp_net_state_pressure COMMAND egp_net_state_pressure_check)
set_tests_properties(egp_net_state_pressure PROPERTIES TIMEOUT 20)
